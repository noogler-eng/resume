% List spacing. Kept even and readable rather than squeezed: itemsep is
% reduced, not zeroed, and topsep is left at its default on purpose --
% the \resumeSubheading macros already end with \vspace{-5pt}, so
% shrinking topsep pulls the first bullet up into the heading line and
% the two print on top of each other. The space for a single page comes
% from the page geometry above instead.
% Note: \resumeItem ends with \vspace{-2pt}, so the visible gap between
% consecutive bullets is itemsep minus 2pt. At itemsep=2pt that came out
% at 11.5pt, tighter than the 12.7pt pitch of a wrapped line inside a
% single bullet, which made continuous bullets run together.
\setlist[itemize,1]{leftmargin=*, parsep=0pt, itemsep=9pt}
\setlist[itemize,2]{parsep=0pt, itemsep=8pt}
